254void ReferenceLineTool::evaluate()
255{
256 try
257 {
258 Calculator calculator;
259 const qreal value = calculator.EvalFormula(m_data->DataVariables(), m_referenceLine.formula);
260 m_formulaValid = qIsFinite(value) && value > 0.0;
261 m_length = m_formulaValid ? ToPixel(value, *m_data->GetPatternUnit()) : 80.0;
262 const QString valueText = qApp->LocaleToString(value) + QLatin1Char(' ') +
263 UnitsToStr(*m_data->GetPatternUnit(), true);
264 m_valueText = m_referenceLine.name.isEmpty() ? valueText
265 : QStringLiteral("%1: %2").arg(m_referenceLine.name, valueText);
266 }
267 catch (...)
268 {
269 m_formulaValid = false;
270 m_length = 80.0;
271 m_valueText = m_referenceLine.name.isEmpty() ? tr("Invalid formula")
272 : tr("%1: invalid formula").arg(m_referenceLine.name);
273 }

276void ReferenceLineTool::updateGeometry()
277{
278 prepareGeometryChange();
279 evaluate();
280
281 QPointF origin(m_referenceLine.xPos, m_referenceLine.yPos);
282 qreal anchoredLineAngle = m_referenceLine.angle;
283 if (m_referenceLine.anchorObject != NULL_ID)
284 {
285 try
286 {
287 const QDomElement line = m_doc->elementById(m_referenceLine.anchorObject, VAbstractPattern::TagLine);
288 if (!line.isNull())
289 {
290 const QPointF first = static_cast<QPointF>(*m_data->GeometricObject<VPointF>(
291 m_doc->GetParametrUInt(line, AttrFirstPoint, NULL_ID_STR)));
292 const QPointF second = static_cast<QPointF>(*m_data->GeometricObject<VPointF>(
293 m_doc->GetParametrUInt(line, AttrSecondPoint, NULL_ID_STR)));
294 origin = first + (second - first) * m_referenceLine.anchorPosition;
295 anchoredLineAngle = QLineF(first, second).angle();
296 }
297 else
298 {
299 origin = static_cast<QPointF>(*m_data->GeometricObject<VPointF>(m_referenceLine.anchorObject));
300 }
301 }
302 catch (...)
303 {
304 m_referenceLine.anchorObject = NULL_ID;
305 }
306 }
307 setPos(origin);
308 qreal displayAngle = m_referenceLine.angle;
309 if (m_referenceLine.orientation == QStringLiteral("horizontal"))
310 {
311 displayAngle = 0.0;
312 }
313 else if (m_referenceLine.orientation == QStringLiteral("vertical"))
314 {
315 displayAngle = 90.0;
316 }
317 else if (m_referenceLine.orientation == QStringLiteral("anchor"))
318 {
319 displayAngle = anchoredLineAngle;
320 }
321 setRotation(displayAngle);
322 setVisible(m_referenceLine.visible);
323 setFlag(QGraphicsItem::ItemIsMovable, m_referenceLine.anchorObject == NULL_ID);
324 setZValue(1000.0);
325 update();
326}
