402void TST_VToolDependency::dependencyDialog()
403{
404 DependencyPattern pattern;
405 pattern.load(patternXml);
406 const Unit unit = Unit::Cm;
407 VContainer data(nullptr, &unit);
408 DependencyTool tool(&pattern, &data, 1);
409 QSignalSpy highlightSpy(&pattern, &VAbstractPattern::ShowTool);
410
411 bool dialogFound = false;
412 int directRows = -1;
413 int recursiveRows = -1;
414 QString firstObject;
415 QString firstType;
416 QString firstReference;
417 QString firstSuggestion;
418 bool openPropertiesShown = false;
419 bool replaceNodeShown = false;
420
421 QTimer::singleShot(0, qApp, [&]()
422 {
423 auto *dialog = qobject_cast<QDialog *>(QApplication::activeModalWidget());
424 dialogFound = dialog != nullptr;
425 if (dialog == nullptr)
426 {
427 return;
428 }
429
430 auto *tree = dialog->findChild<QTreeWidget *>();
431 auto *recursive = dialog->findChild<QCheckBox *>();
432 if (tree != nullptr)
433 {
434 directRows = tree->topLevelItemCount();
435 if (directRows > 0)
436 {
437 firstObject = tree->topLevelItem(0)->text(0);
438 firstType = tree->topLevelItem(0)->text(1);
439 firstReference = tree->topLevelItem(0)->text(2);
440 firstSuggestion = tree->topLevelItem(0)->text(4);
441 tree->itemClicked(tree->topLevelItem(0), 0);
442 auto *openButton = dialog->findChild<QPushButton *>(
443 QStringLiteral("dependencyOpenPropertiesButton"));
444 auto *replaceButton = dialog->findChild<QPushButton *>(
445 QStringLiteral("dependencyReplaceNodeButton"));
446 openPropertiesShown = openButton != nullptr && !openButton->isHidden();
447 replaceNodeShown = replaceButton != nullptr && !replaceButton->isHidden();
448 }
449 }
450 if (recursive != nullptr)
451 {
452 recursive->setChecked(true);
453 recursiveRows = tree == nullptr ? -1 : tree->topLevelItemCount();
454 }
455 dialog->accept();
456 });
457
458 tool.showDependencies();
459
460 QVERIFY(dialogFound);
461 QCOMPARE(directRows, 1);
462 QCOMPARE(recursiveRows, 2);
463 QCOMPARE(firstObject, QStringLiteral("B"));
464 QVERIFY(!firstType.isEmpty());
465 QVERIFY(!firstReference.isEmpty());
466 QCOMPARE(firstSuggestion, QStringLiteral("Select the object and replace base point in Properties"));
467 QVERIFY(openPropertiesShown);
468 QVERIFY(!replaceNodeShown);
469 QVERIFY(!highlightSpy.isEmpty());
470 QCOMPARE(highlightSpy.at(0).at(0).toUInt(), quint32(2));
471 QCOMPARE(highlightSpy.at(0).at(1).toBool(), true);
