65/// @brief childElements serialize the child elements of a tool element.
66///
67/// Some tools keep object references in child elements instead of attributes, for example the points of a
68/// spline path. Text nodes are skipped.
69///
70/// @param element tool element.
71/// @return the child elements as one string.
72//---------------------------------------------------------------------------------------------------------------------
73
74QString childElements(const QDomElement &element)
75{
76 QString dump;
77 QTextStream stream(&dump);
78
79 QDomElement child = element.firstChildElement();
80 while (!child.isNull())
81 {
82 child.save(stream, 0);
83 child = child.nextSiblingElement();
84 }
85 return dump;
86}

88//---------------------------------------------------------------------------------------------------------------------
89/// @brief referencesChanged check if a tool option change replaced a referenced object.
90///
91/// The reference counters that guard deletion are only set when a tool is created during a full parse.
92/// When an option change replaces a referenced object the counters become stale, so the caller has to
93/// request a full parse to rebuild them.
94///
95/// @param oldXml tool element before the change.
96/// @param newXml tool element after the change.
97/// @return true if an object reference changed.
98//---------------------------------------------------------------------------------------------------------------------
99
100bool referencesChanged(const QDomElement &oldXml, const QDomElement &newXml)
101{
102 const QSet<QString> &referenceAttributes = VAbstractPattern::objectReferenceAttributes();
103 for (auto attribute = referenceAttributes.constBegin(); attribute != referenceAttributes.constEnd(); ++attribute)
104 {
105 if (oldXml.attribute(*attribute) != newXml.attribute(*attribute))
106 {
107 return true;
108 }
109 }
110
111 return childElements(oldXml) != childElements(newXml);
112}

129void SaveToolOptions::undo()
130{
131 qCDebug(vUndo, "Undo.");
132
133 QDomElement domElement = doc->elementById(nodeId);
134 if (domElement.isElement())
135 {
136 domElement.parentNode().replaceChild(oldXml, domElement);
137
138 if (referencesChanged(oldXml, newXml))
139 {
140 // A referenced object changed. Only a full parse rebuilds the reference counters.
141 emit NeedFullParsing();
142 }
143 else
144 {
145 emit NeedLiteParsing(Document::LiteParse);
146 }
147 }
148 else
149 {
150 qCWarning(vUndo, "Can't find tool with id = %u.", nodeId);
151 return;
152 }
153}

183bool SaveToolOptions::mergeWith(const QUndoCommand *command)
184{
185 const SaveToolOptions *saveCommand = static_cast<const SaveToolOptions *>(command);
186 SCASSERT(saveCommand != nullptr)
187 const quint32 id = saveCommand->getToolId();
188
189 if (id != nodeId)
190 {
191 return false;
192 }
193
194 newXml = saveCommand->getNewXml();
195 return true;
196}
