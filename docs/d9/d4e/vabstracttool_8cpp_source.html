177/**
178 * @brief CheckFormula check formula.
179 *
180 * Try calculate formula. If find error show dialog that allow user try fix formula. If user can't throw exception. In
181 * successes case return result calculation and fixed formula string. If formula ok don't touch formula.
182 *
183 * @param toolId [in] tool's id.
184 * @param formula [in|out] string with formula.
185 * @param data [in] container with variables. Need for math parser.
186 * @throw QmuParserError.
187 * @return result of calculation formula.
188 */
189qreal VAbstractTool::CheckFormula(const quint32 &toolId, QString &formula, VContainer *data)
190{
191 SCASSERT(data != nullptr)
192 qreal result = 0;
193 try
194 {
195 QScopedPointer<Calculator> cal(new Calculator());
196 result = cal->EvalFormula(data->DataVariables(), formula);
197
198 if (qIsInf(result) || qIsNaN(result))
199 {
200 qWarning() << "Invalid the formula value";
201 return 0;
202 }
203 }
204 catch (qmu::QmuParserError &error)
205 {
206 qDebug() << "\nMath parser error:\n"
207 << "--------------------------------------\n"
208 << "Message: " << error.GetMsg() << "\n"
209 << "Expression: " << error.GetExpr() << "\n"
210 << "--------------------------------------";
211
212 if (qApp->isAppInGUIMode())
213 {
214 QScopedPointer<DialogUndo> dialogUndo(new DialogUndo(qApp->getMainWindow()));
215 forever
216 {
217 if (dialogUndo->exec() == QDialog::Accepted)
218 {
219 const UndoButton resultUndo = dialogUndo->Result();
220 if (resultUndo == UndoButton::Fix)
221 {
222 auto *dialog = new EditFormulaDialog(data, toolId, ToolDialog, qApp->getMainWindow());
223 dialog->setWindowTitle(tr("Edit wrong formula"));
224 dialog->SetFormula(formula);
225 if (dialog->exec() == QDialog::Accepted)
226 {
227 formula = dialog->GetFormula();
228 /* Need delete dialog here because parser in dialog don't allow use correct separator for
229 * parsing here. */
230 delete dialog;
231 QScopedPointer<Calculator> cal1(new Calculator());
232 result = cal1->EvalFormula(data->DataVariables(), formula);
233
234 if (qIsInf(result) || qIsNaN(result))
235 {
236 qWarning() << "Invalid the formula value";
237 return 0;
238 }
239
240 break;
241 }
242 else
243 {
244 delete dialog;
245 }
246 }
247 else
248 {
249 throw VExceptionUndo(QString("Undo wrong formula %1").arg(formula));
250 }
251 }
252 else
253 {
254 throw;
255 }
256 }
257 }
258 else
259 {
260 throw;
261 }
262 }
263 return result;
264}

268 * @brief deleteTool full delete object from scene and file.
269 */
270void VAbstractTool::deleteTool(bool ask)
271{
272 qCDebug(vTool, "Deleting abstract tool.");
273 if (not isUsed())
274 {
275 qCDebug(vTool, "No children.");
276 qApp->getSceneView()->itemClicked(nullptr);
277 if (ask)
278 {
279 qCDebug(vTool, "Asking.");
280 if (ConfirmDeletion() == QMessageBox::No)
281 {
282 qCDebug(vTool, "User said no.");
283 return;
284 }
285 }
286
287 qCDebug(vTool, "Begin deleting.");
288 DelTool *delTool = new DelTool(doc, m_id);
289 connect(delTool, &DelTool::NeedFullParsing, doc, &VAbstractPattern::NeedFullParsing);
290 qApp->getUndoStack()->push(delTool);
291
292 // Throw exception, this will help prevent case when we forget to immediately quit function.
293 VExceptionToolWasDeleted e("Tool was used after deleting.");
294 throw e;
295 }
296 else
297 {
298 qCWarning(vTool, "Can't delete, tool has children.");
299 showDependencies();
300 }
301}

304void VAbstractTool::showDependencies()
305{
306 enum class DependencyAction
307 {
308 None,
309 Select,
310 EditPiece,
311 ReplaceNode,
312 RemoveNode
313 };
314
315 QDialog dialog(qApp->getMainWindow());
316 dialog.setWindowTitle(tr("Dependencies"));
317 dialog.resize(720, 360);
318
319 auto *layout = new QVBoxLayout(&dialog);
320 QString objectName = tr("Object %1").arg(m_id);
321 try
322 {
323 objectName = getData()->GetGObject(m_id)->name();
324 }
325 catch (const VExceptionBadId &)
326 {
327 try
328 {
329 objectName = getData()->GetPiece(m_id).GetName();
330 }
331 catch (const VExceptionBadId &)
332 {
333 // Some tools, for example lines and operations, have no object stored under the tool id.
334 }
335 }
336 auto *label = new QLabel(&dialog);
337 label->setWordWrap(true);
338 layout->addWidget(label);
339
340 auto *recursive = new QCheckBox(tr("Show all descendants"), &dialog);
341 layout->addWidget(recursive);
342
343 auto *tree = new QTreeWidget(&dialog);
344 tree->setColumnCount(5);
345 tree->setHeaderLabels(QStringList() << tr("Object") << tr("Type") << tr("Dependency") << tr("Draft block")
346 << tr("Suggested action"));
347 tree->setRootIsDecorated(false);
348 layout->addWidget(tree);
349
350 quint32 selectedToolId = NULL_ID;
351 Tool selectedToolType = Tool::LAST_ONE_DO_NOT_USE;
352 DependencyAction selectedAction = DependencyAction::None;
353
354 auto *actionLabel = new QLabel(tr("Select an object to see the available next steps."), &dialog);
355 actionLabel->setWordWrap(true);
356 layout->addWidget(actionLabel);
357
358 auto *actionLayout = new QHBoxLayout;
359 auto *selectButton = new QPushButton(tr("Open properties"), &dialog);
360 auto *replaceButton = new QPushButton(tr("Replace contour section..."), &dialog);
361 auto *removeButton = new QPushButton(tr("Remove from pattern piece..."), &dialog);
362 auto *editPieceButton = new QPushButton(tr("Edit pattern piece"), &dialog);
363 selectButton->setObjectName(QStringLiteral("dependencyOpenPropertiesButton"));
364 replaceButton->setObjectName(QStringLiteral("dependencyReplaceNodeButton"));
365 removeButton->setObjectName(QStringLiteral("dependencyRemoveNodeButton"));
366 editPieceButton->setObjectName(QStringLiteral("dependencyEditPieceButton"));
367 actionLayout->addWidget(selectButton);
368 actionLayout->addWidget(replaceButton);
369 actionLayout->addWidget(removeButton);
370 actionLayout->addWidget(editPieceButton);
371 actionLayout->addStretch();
372 layout->addLayout(actionLayout);
373
374 selectButton->hide();
375 replaceButton->hide();
376 removeButton->hide();
377 editPieceButton->hide();
378
379 auto *buttons = new QDialogButtonBox(QDialogButtonBox::Close, &dialog);
380
381 auto clearActions = [this, actionLabel, selectButton, replaceButton, removeButton, editPieceButton]()
382 {
383 actionLabel->setText(tr("Select an object to see the available next steps."));
384 selectButton->hide();
385 replaceButton->hide();
386 removeButton->hide();
387 editPieceButton->hide();
388 };
389
390 auto populate = [this, tree, recursive, label, objectName, clearActions]()
391 {
392 tree->clear();
393 clearActions();
394 const QVector<VToolDependency> dependencies = recursive->isChecked()
395 ? doc->getDependentObjectsRecursive(m_id, getData())
396 : doc->getDirectDependencies(m_id, getData());
397 label->setText(dependencies.isEmpty()
398 ? tr("No dependent objects were found for %1.").arg(objectName)
399 : tr("%1 cannot be deleted because these objects depend on it.").arg(objectName));
400 for (const VToolDependency &dependency : dependencies)
401 {
402 const QString indentation(qMax(0, dependency.depth - 1) * 3, QLatin1Char(' '));
403 QString suggestion = tr("Select the object and replace %1 in Properties").arg(dependency.reference);
404 if (dependency.id == NULL_ID)
405 {
406 suggestion = tr("Edit the formula");
407 }
408 else if (dependency.type == Tool::Piece)
409 {
410 suggestion = tr("Edit the pattern piece");
411 }
412 else if (dependency.type == Tool::NodePoint || dependency.type == Tool::NodeArc ||
413 dependency.type == Tool::NodeElArc || dependency.type == Tool::NodeSpline ||
414 dependency.type == Tool::NodeSplinePath)
415 {
416 suggestion = tr("Replace or remove the node in the pattern piece");
417 }
418 auto *item = new QTreeWidgetItem(tree, QStringList() << indentation + dependency.name
419 << dependency.typeName << dependency.reference
420 << dependency.draftBlockName << suggestion);
421 item->setData(0, Qt::UserRole, dependency.id);
422 item->setData(0, Qt::UserRole + 1, static_cast<int>(dependency.type));
423 }
424 tree->resizeColumnToContents(0);
425 tree->resizeColumnToContents(1);
426 tree->resizeColumnToContents(2);
427 };
428
429 connect(recursive, &QCheckBox::toggled, &dialog, populate);
430 connect(tree, &QTreeWidget::itemClicked, &dialog,
431 [this, tree, actionLabel, selectButton, replaceButton, removeButton, editPieceButton,
432 &selectedToolId, &selectedToolType](QTreeWidgetItem *item)
433 {
434 selectedToolId = item->data(0, Qt::UserRole).toUInt();
435 selectedToolType = static_cast<Tool>(item->data(0, Qt::UserRole + 1).toInt());
436 const bool isNode = selectedToolType == Tool::NodePoint || selectedToolType == Tool::NodeArc ||
437 selectedToolType == Tool::NodeElArc || selectedToolType == Tool::NodeSpline ||
438 selectedToolType == Tool::NodeSplinePath;
439
440 selectButton->setVisible(selectedToolId != NULL_ID && !isNode && selectedToolType != Tool::Piece);
441 replaceButton->setVisible(isNode);
442 removeButton->setVisible(isNode);
443 editPieceButton->setVisible(isNode || selectedToolType == Tool::Piece);
444 actionLabel->setText(isNode
445 ? tr("This node or an adjacent section can be replaced by one or more points and curves. Changes can be undone.")
446 : selectedToolType == Tool::Piece
447 ? tr("Open the pattern piece to change the references listed above.")
448 : selectedToolId == NULL_ID
449 ? tr("Open the formula editor for this variable and replace the reference there.")
450 : tr("Open Properties and replace the referenced object."));
451 for (int i = 0; i < tree->topLevelItemCount(); ++i)
452 {
453 const quint32 id = tree->topLevelItem(i)->data(0, Qt::UserRole).toUInt();
454 if (id != NULL_ID)
455 {
456 emit doc->ShowTool(id, tree->topLevelItem(i) == item);
457 }
458 }
459 });
460
461 connect(selectButton, &QPushButton::clicked, &dialog, [&dialog, &selectedAction]()
462 {
463 selectedAction = DependencyAction::Select;
464 dialog.accept();
465 });
466 connect(replaceButton, &QPushButton::clicked, &dialog, [&dialog, &selectedAction]()
467 {
468 selectedAction = DependencyAction::ReplaceNode;
469 dialog.accept();
470 });
471 connect(removeButton, &QPushButton::clicked, &dialog, [&dialog, &selectedAction]()
472 {
473 selectedAction = DependencyAction::RemoveNode;
474 dialog.accept();
475 });
476 connect(editPieceButton, &QPushButton::clicked, &dialog, [&dialog, &selectedAction]()
477 {
478 selectedAction = DependencyAction::EditPiece;
479 dialog.accept();
480 });
481 connect(buttons, &QDialogButtonBox::rejected, &dialog, &QDialog::reject);
482 layout->addWidget(buttons);
483
484 populate();
485 dialog.exec();
486 const QVector<VToolDependency> dependencies = doc->getDependentObjectsRecursive(m_id, getData());
487 for (const VToolDependency &dependency : dependencies)
488 {
489 if (dependency.id != NULL_ID)
490 {
491 emit doc->ShowTool(dependency.id, false);
492 }
493 }
494
495 quint32 selectedPathId = NULL_ID;
496 auto findPieceTool = [this, selectedToolId, selectedToolType, &selectedPathId]() -> PatternPieceTool *
497 {
498 quint32 pieceId = selectedToolType == Tool::Piece ? selectedToolId : NULL_ID;
499 if (pieceId == NULL_ID)
500 {
501 const QVector<VToolDependency> dependencies = doc->getDirectDependencies(selectedToolId, getData());
502 for (const VToolDependency &dependency : dependencies)
503 {
504 if (dependency.type == Tool::Piece)
505 {
506 pieceId = dependency.id;
507 break;
508 }
509 if (dependency.type == Tool::InternalPath)
510 {
511 selectedPathId = dependency.id;
512 pieceId = getData()->pieceIdOfPath(selectedPathId);
513 break;
514 }
515 }
516 }
517
518 if (pieceId == NULL_ID)
519 {
520 return nullptr;
521 }
522 try
523 {
524 return qobject_cast<PatternPieceTool *>(VAbstractPattern::getTool(pieceId));
525 }
526 catch (const VExceptionBadId &)
527 {
528 return nullptr;
529 }
530 };
531
532 if (selectedAction == DependencyAction::ReplaceNode || selectedAction == DependencyAction::RemoveNode ||
533 selectedAction == DependencyAction::EditPiece)
534 {
535 PatternPieceTool *pieceTool = findPieceTool();
536 if (pieceTool == nullptr)
537 {
538 QMessageBox::information(qApp->getMainWindow(), tr("Dependencies"),
539 tr("The pattern piece is in another draft block. Open that draft block and try again."));
540 return;
541 }
542
543 if (selectedAction == DependencyAction::ReplaceNode || selectedAction == DependencyAction::RemoveNode)
544 {
545 pieceTool->replacePieceNode(selectedToolId, selectedPathId,
546 selectedAction == DependencyAction::RemoveNode);
547 QTimer::singleShot(0, this, [this]() { showDependencies(); });
548 return;
549 }
550
551 connect(pieceTool, &PatternPieceTool::piecePropertiesClosed, this, [this, pieceTool]()
552 {
553 disconnect(pieceTool, &PatternPieceTool::piecePropertiesClosed, this, nullptr);
554 QTimer::singleShot(0, this, [this]() { showDependencies(); });
555 });
556 pieceTool->editPieceProperties(selectedToolType == Tool::Piece ? NULL_ID : selectedToolId);
557 return;
558 }
559
560 if (selectedAction == DependencyAction::Select && selectedToolId != NULL_ID)
561 {
562 try
563 {
564 auto *item = dynamic_cast<QGraphicsItem *>(VAbstractPattern::getTool(selectedToolId));
565 if (item != nullptr)
566 {
567 item->setSelected(true);
568 qApp->getSceneView()->itemClicked(item);
569 }
570 }
571 catch (const VExceptionBadId &)
572 {
573 // A dependency from another draft block may not be loaded in the current scene.
574 }
575 }
576}

579int VAbstractTool::ConfirmDeletion()
580{
581 if (false == qApp->Settings()->getConfirmItemDelete())
582 {
583 return QMessageBox::Yes;
584 }
585
586 Utils::CheckableMessageBox msgBox(qApp->getMainWindow());
587 msgBox.setWindowTitle(tr("Confirm deletion"));
588 msgBox.setText(tr("Do you really want to delete?"));
589 msgBox.setStandardButtons(QDialogButtonBox::Yes | QDialogButtonBox::No);
590 msgBox.setDefaultButton(QDialogButtonBox::No);
591 msgBox.setIconPixmap(QApplication::style()->standardIcon(QStyle::SP_MessageBoxQuestion).pixmap(32, 32) );
592
593 int dialogResult = msgBox.exec();
594
595 if (dialogResult == QDialog::Accepted)
596 {
597 qApp->Settings()->setConfirmItemDelete(not msgBox.isChecked());
598 }
599
600 return dialogResult == QDialog::Accepted ? QMessageBox::Yes : QMessageBox::No;
601}

901//---------------------------------------------------------------------------------------------------------------------
902QDomElement VAbstractTool::AddSANode(VAbstractPattern *doc, const QString &tagName, const VPieceNode &node)
903{
904 QDomElement nod = doc->createElement(tagName);
905
906 doc->SetAttribute(nod, AttrIdObject, node.GetId());
907
908 const Tool type = node.GetTypeTool();
909 if (type != Tool::NodePoint)
910 {
911 doc->SetAttribute(nod, VAbstractPattern::AttrNodeReverse, static_cast<quint8>(node.GetReverse()));
912 }
913 else
914 {
915 if (node.GetFormulaSABefore() != currentSeamAllowance)
916 {
917 doc->SetAttribute(nod, VAbstractPattern::AttrSABefore, node.GetFormulaSABefore());
918 }
919
920 if (node.GetFormulaSAAfter() != currentSeamAllowance)
921 {
922 doc->SetAttribute(nod, VAbstractPattern::AttrSAAfter, node.GetFormulaSAAfter());
923 }
924 }
925
926 {
927 const bool excluded = node.isExcluded();
928 if (excluded)
929 {
930 doc->SetAttribute(nod, VAbstractPattern::AttrNodeExcluded, excluded);
931 }
932 else
933 { // For backward compatebility.
934 nod.removeAttribute(VAbstractPattern::AttrNodeExcluded);
935 }
936 }
937
938 switch (type)
939 {
940 case (Tool::NodeArc):
941 doc->SetAttribute(nod, AttrType, VAbstractPattern::NodeArc);
942 break;
943 case (Tool::NodeElArc):
944 doc->SetAttribute(nod, AttrType, VAbstractPattern::NodeElArc);
945 break;
946 case (Tool::NodePoint):
947 doc->SetAttribute(nod, AttrType, VAbstractPattern::NodePoint);
948 break;
949 case (Tool::NodeSpline):
950 doc->SetAttribute(nod, AttrType, VAbstractPattern::NodeSpline);
951 break;
952 case (Tool::NodeSplinePath):
953 doc->SetAttribute(nod, AttrType, VAbstractPattern::NodeSplinePath);
954 break;
955 default:
956 qWarning() << "May be wrong tool type!!! Ignoring."<<Q_FUNC_INFO;
957 break;
958 }
959
960 {
961 const unsigned char angleType = static_cast<unsigned char>(node.GetAngleType());
962
963 if (angleType > 0)
964 {
965 doc->SetAttribute(nod, AttrAngle, angleType);
966 }
967 }
968
969 if (type == Tool::NodePoint)
970 {
971 if (node.isNotch())
972 {
973 doc->SetAttribute(nod, VAbstractPattern::AttrNodeIsNotch, node.isNotch());
974 doc->SetAttribute(nod, VAbstractPattern::AttrNodeShowNotch, node.showNotch());
975 doc->SetAttribute(nod, VAbstractPattern::AttrNodeShowSecondNotch, node.showSeamlineNotch());
976 doc->SetAttribute(nod, VAbstractPattern::AttrNodeNotchType, notchTypeToString(node.getNotchType()));
977 doc->SetAttribute(nod, VAbstractPattern::AttrNodeNotchSubType, notchSubTypeToString(node.getNotchSubType()));
978 doc->SetAttribute(nod, VAbstractPattern::AttrNodeNotchLength, node.getNotchLength());
979 doc->SetAttribute(nod, VAbstractPattern::AttrNodeNotchWidth, node.getNotchWidth());
980 doc->SetAttribute(nod, VAbstractPattern::AttrNodeNotchAngle, node.getNotchAngle());
981 doc->SetAttribute(nod, VAbstractPattern::AttrNodeNotchCount, node.getNotchCount());
982 }
983
984 if (not node.isNotch() && node.getNotchType() == NotchType::Slit
985 && node.getNotchSubType() == NotchSubType::Straightforward)
986 { // For backward compatebility.
987 nod.removeAttribute(VAbstractPattern::AttrNodeIsNotch);
988 nod.removeAttribute(VAbstractPattern::AttrNodeNotchType);
989 nod.removeAttribute(VAbstractPattern::AttrNodeNotchSubType);
990 }
991 }
992 else
993 { // Wrong configuration.
994 nod.removeAttribute(VAbstractPattern::AttrNodeIsNotch);
995 nod.removeAttribute(VAbstractPattern::AttrNodeNotchType);
996 nod.removeAttribute(VAbstractPattern::AttrNodeNotchSubType);
997 }
998
999 return nod;
1000}
