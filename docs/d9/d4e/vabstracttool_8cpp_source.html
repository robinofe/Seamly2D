162QString ReplacementDiagnostics(VAbstractPattern *doc, const VContainer *data,
163 const QSharedPointer<VGObject> &source, quint32 sourceObjectId,
164 const QSet<quint32> &trackedObjects, bool allowTrackedPointDetachment,
165 bool requireTrackedObject)
166{
167 QString report;
168 QTextStream out(&report);
169 out << "Seamly2D replacement diagnostics v1\n";
170 out << "Diagnostic build marker: replacement-diagnostics-2026-08-17\n";
171 out << "Executable: " << QCoreApplication::applicationFilePath() << '\n';
172 out << "Application version: " << QCoreApplication::applicationVersion() << '\n';
173 out << "Qt version: " << qVersion() << "\n\n";
174 out << "Source object ID: " << sourceObjectId << '\n';
175 out << "Source name: " << source->name() << '\n';
176 out << "Source geometry type: " << GeometryTypeName(source->getType()) << " ("
177 << static_cast<int>(source->getType()) << ")\n";
178 out << "Source tool ID: " << source->getIdTool() << "\n\n";
179
180 QSet<quint32> descendantTools;
181 const QVector<VToolDependency> descendants =
182 doc->getDependentObjectsRecursive(source->getIdTool(), data);
183 out << "Dependency descendants: " << descendants.size() << '\n';
184 for (const VToolDependency &dependency : descendants)
185 {
186 if (dependency.id != NULL_ID)
187 {
188 descendantTools.insert(dependency.id);
189 }
190 out << " tool=" << dependency.id << ", depth=" << dependency.depth
191 << ", name=" << dependency.name << ", type=" << dependency.typeName
192 << ", reference=" << dependency.reference << '\n';
193 }
194
195 const auto geometry = data->DataGObjects();
196 out << "\nGeometry object count: " << geometry->size() << '\n';
197 out << "Tracked/new object IDs:";
198 for (quint32 id : trackedObjects)
199 {
200 out << ' ' << id;
201 }
202 out << "\n\nCandidate evaluation:\n";
203
204 bool foundRelevantObject = false;
205 for (auto object = geometry->constBegin(); object != geometry->constEnd(); ++object)
206 {
207 const QSharedPointer<VGObject> candidate = object.value();
208 const bool tracked = trackedObjects.contains(object.key());
209 const bool sameType = candidate->getType() == source->getType();
210 if (!tracked && !sameType)
211 {
212 continue;
213 }
214 foundRelevantObject = true;
215 const bool descendant = descendantTools.contains(candidate->getIdTool());
216 const bool canDetach = allowTrackedPointDetachment && tracked && source->getType() == GOType::Point;
217 QString result = QStringLiteral("eligible");
218 if (candidate->getMode() != Draw::Calculation)
219 {
220 result = QStringLiteral("excluded: mode is not Calculation");
221 }
222 else if (candidate->getIdObject() != NULL_ID)
223 {
224 result = QStringLiteral("excluded: idObject is not 0");
225 }
226 else if (candidate->getIdTool() == source->getIdTool())
227 {
228 result = QStringLiteral("excluded: same tool as source");
229 }
230 else if (requireTrackedObject && !tracked)
231 {
232 result = QStringLiteral("excluded: object existed before replacement session");
233 }
234 else if (!sameType)
235 {
236 result = QStringLiteral("excluded: geometry type differs from source");
237 }
238 else if (descendant && !canDetach)
239 {
240 result = QStringLiteral("excluded: descendant of source");
241 }
242 else if (descendant)
243 {
244 result = QStringLiteral("eligible: tracked point will be detached");
245 }
246
247 out << "- object ID: " << object.key() << '\n';
248 out << " name: " << candidate->name() << '\n';
249 out << " geometry type: " << GeometryTypeName(candidate->getType()) << " ("
250 << static_cast<int>(candidate->getType()) << ")\n";
251 out << " tool ID: " << candidate->getIdTool() << '\n';
252 out << " mode: " << DrawModeName(candidate->getMode()) << " ("
253 << static_cast<int>(candidate->getMode()) << ")\n";
254 out << " idObject: " << candidate->getIdObject() << '\n';
255 out << " tracked/new: " << (tracked ? "yes" : "no") << '\n';
256 out << " descendant: " << (descendant ? "yes" : "no") << '\n';
257 out << " result: " << result << '\n';
258
259 const QDomElement element = doc->elementById(candidate->getIdTool());
260 if (element.isNull())
261 {
262 out << " XML: <not found>\n";
263 }
264 else
265 {
266 QDomDocument snippet;
267 snippet.appendChild(snippet.importNode(element, true));
268 QString xml = snippet.toString(-1).trimmed();
269 xml.replace('\n', ' ');
270 out << " XML: " << xml << '\n';
271 }
272 }
273 if (!foundRelevantObject)
274 {
275 out << " <no tracked or type-compatible geometry objects found>\n";
276 }
277 return report;
278}

338/**
339 * @brief CheckFormula check formula.
340 *
341 * Try calculate formula. If find error show dialog that allow user try fix formula. If user can't throw exception. In
342 * successes case return result calculation and fixed formula string. If formula ok don't touch formula.
343 *
344 * @param toolId [in] tool's id.
345 * @param formula [in|out] string with formula.
346 * @param data [in] container with variables. Need for math parser.
347 * @throw QmuParserError.
348 * @return result of calculation formula.
349 */
350qreal VAbstractTool::CheckFormula(const quint32 &toolId, QString &formula, VContainer *data)
351{
352 SCASSERT(data != nullptr)
353 qreal result = 0;
354 try
355 {
356 QScopedPointer<Calculator> cal(new Calculator());
357 result = cal->EvalFormula(data->DataVariables(), formula);
358
359 if (qIsInf(result) || qIsNaN(result))
360 {
361 qWarning() << "Invalid the formula value";
362 return 0;
363 }
364 }
365 catch (qmu::QmuParserError &error)
366 {
367 qDebug() << "\nMath parser error:\n"
368 << "--------------------------------------\n"
369 << "Message: " << error.GetMsg() << "\n"
370 << "Expression: " << error.GetExpr() << "\n"
371 << "--------------------------------------";
372
373 if (qApp->isAppInGUIMode())
374 {
375 QScopedPointer<DialogUndo> dialogUndo(new DialogUndo(qApp->getMainWindow()));
376 forever
377 {
378 if (dialogUndo->exec() == QDialog::Accepted)
379 {
380 const UndoButton resultUndo = dialogUndo->Result();
381 if (resultUndo == UndoButton::Fix)
382 {
383 auto *dialog = new EditFormulaDialog(data, toolId, ToolDialog, qApp->getMainWindow());
384 dialog->setWindowTitle(tr("Edit wrong formula"));
385 dialog->SetFormula(formula);
386 if (dialog->exec() == QDialog::Accepted)
387 {
388 formula = dialog->GetFormula();
389 /* Need delete dialog here because parser in dialog don't allow use correct separator for
390 * parsing here. */
391 delete dialog;
392 QScopedPointer<Calculator> cal1(new Calculator());
393 result = cal1->EvalFormula(data->DataVariables(), formula);
394
395 if (qIsInf(result) || qIsNaN(result))
396 {
397 qWarning() << "Invalid the formula value";
398 return 0;
399 }
400
401 break;
402 }
403 else
404 {
405 delete dialog;
406 }
407 }
408 else
409 {
410 throw VExceptionUndo(QString("Undo wrong formula %1").arg(formula));
411 }
412 }
413 else
414 {
415 throw;
416 }
417 }
418 }
419 else
420 {
421 throw;
422 }
423 }
424 return result;
425}

431void VAbstractTool::deleteTool(bool ask)
432{
433 qCDebug(vTool, "Deleting abstract tool.");
434 if (not isUsed())
435 {
436 qCDebug(vTool, "No children.");
437 qApp->getSceneView()->itemClicked(nullptr);
438 if (ask)
439 {
440 qCDebug(vTool, "Asking.");
441 if (ConfirmDeletion() == QMessageBox::No)
442 {
443 qCDebug(vTool, "User said no.");
444 return;
445 }
446 }
447
448 qCDebug(vTool, "Begin deleting.");
449 DelTool *delTool = new DelTool(doc, m_id);
450 connect(delTool, &DelTool::NeedFullParsing, doc, &VAbstractPattern::NeedFullParsing);
451 qApp->getUndoStack()->push(delTool);
452
453 // Throw exception, this will help prevent case when we forget to immediately quit function.
454 VExceptionToolWasDeleted e("Tool was used after deleting.");
455 throw e;
456 }
457 else
458 {
459 // The dependency dialog contains the reason and the available repair actions. Avoid a separate warning first.
460 showDependencies();
461 }
462}
